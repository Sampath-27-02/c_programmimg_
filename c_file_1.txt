#include <stdio.h>
#include<string.h>
int main() {
    int num;
    int n=0,i;
    char arr[10][10]={"zero","one","two","three","four","five","six","seven","eight","nine"};
    printf("Enter a number: ");
    scanf("%d",&num);
    int temp=num;
    while(temp!=0){
        temp/=10;
        n++;
    }
    char a[n][10];
    temp=num;
    for(i=0;i<n && temp!=0;i++){
        int rem=temp%10;
        strcpy(a[i],arr[rem]);
        temp/=10;
    }
    for(i=n-1;i>=0;i--){
        printf("%s ",a[i]);
    }
    return 0;
}
