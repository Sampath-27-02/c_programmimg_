#include <stdio.h>
#include<math.h>
int main() {
    while(1){
        int num;
        printf("Enter a number: ");
        scanf("%d",&num);
        if(num<0){
            printf("Negative Skipped\n");
            continue;
            
        }
        else if(num==999){
            printf("Exits program if number is 999");
            return 0;}
        else
            printf("Square root of the given number is %.2f\n",pow(num,.5));
    }
    return 0;
}
